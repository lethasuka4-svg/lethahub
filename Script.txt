-- SÜRÜKLEME SİSTEMİ (MOBİL UYUMLU)
local function makeDraggable(guiObject)
    local dragging, hasDragged, dragStart, startPos = false, false, nil, nil
    guiObject.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            hasDragged = false
            dragStart = input.Position
            startPos = guiObject.Position
            local conn
            conn = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    conn:Disconnect()
                end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            if delta.Magnitude > 6 then hasDragged = true end
            guiObject.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
    return function() return hasDragged end
end

-- CAMLOCK SİSTEMİ
local function getClosestTarget()
    local nearest = nil
    local shortestDistance = math.huge
    local cam = workspace.CurrentCamera
    if not cam then return nil end
    local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local tHrp = player.Character:FindFirstChild("HumanoidRootPart")
            local tHum = player.Character:FindFirstChildOfClass("Humanoid")
            if tHrp and tHum and tHum.Health > 0 then
                local pos, onScreen = cam:WorldToViewportPoint(tHrp.Position)
                if onScreen then
                    local dist = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                    if dist < shortestDistance then
                        shortestDistance = dist
                        nearest = player
                    end
                end
            end
        end
    end
    return nearest
end

local function toggleCamlock()
    camlockActive = not camlockActive
    if camlockActive then
        camlockTarget = getClosestTarget()
        LockUI.Text = "Lock: ON"
        LockUI.TextColor3 = Color3.fromRGB(0, 240, 255)
        LockStroke.Color = Color3.fromRGB(0, 240, 255)
        CamlockMenuBtn.Text = "Camlock: ON"
        CamlockMenuBtn.BackgroundColor3 = Color3.fromRGB(0, 160, 90)
    else
        camlockTarget = nil
        LockUI.Text = "Lock: OFF"
        LockUI.TextColor3 = Color3.fromRGB(255, 60, 80)
        LockStroke.Color = Color3.fromRGB(255, 60, 80)
        CamlockMenuBtn.Text = "Camlock: OFF"
        CamlockMenuBtn.BackgroundColor3 = Color3.fromRGB(160, 35, 50)
    end
end

-- ANA DÖNGÜ (RENDER STEPPED)
RunService.RenderStepped:Connect(function(delta)
    if LocalPlayer.Character then
        local char = LocalPlayer.Character
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        local hrp = char:FindFirstChild("HumanoidRootPart")

        -- 1. CAMLOCK VE HAREKET TAHMİNİ
        if camlockActive then
            if not camlockTarget or not camlockTarget.Character or not camlockTarget.Character:FindFirstChild("HumanoidRootPart") or (camlockTarget.Character:FindFirstChildOfClass("Humanoid") and camlockTarget.Character:FindFirstChildOfClass("Humanoid").Health <= 0) then
                camlockTarget = getClosestTarget()
            end

            if camlockTarget and camlockTarget.Character and camlockTarget.Character:FindFirstChild("HumanoidRootPart") then
                local tHrp = camlockTarget.Character.HumanoidRootPart
                local predictedPos = tHrp.Position + (tHrp.AssemblyLinearVelocity * predictionAmount)
                local cam = workspace.CurrentCamera
                if cam then
                    cam.CFrame = CFrame.new(cam.CFrame.Position, predictedPos)
                end
            end
        end

        -- 2. DÜZELTİLMİŞ GÖRÜNMEZ ZEMİN (ALT PLATFORMA İNMEYİ SAĞLAR)
        if voidFloorActive and hrp and humanoid then
            local rayParams = RaycastParams.new()
            rayParams.FilterDescendantsInstances = {char, voidPlatform}
            rayParams.FilterType = Enum.RaycastFilterType.Exclude

            local rayResult = workspace:Raycast(hrp.Position, Vector3.new(0, -1000, 0), rayParams)
            if rayResult then
                lastGroundY = rayResult.Position.Y - 2.5
            end

            if lastGroundY ~= 0 then
                voidPlatform.CFrame = CFrame.new(hrp.Position.X, lastGroundY, hrp.Position.Z)
                voidPlatform.CanCollide = true
            end
        else
            voidPlatform.CanCollide = false
        end

        -- 3. MOBİL ZIPLAMA BUTONU DÜZELTME
        if infJumpActive then
            pcall(function()
                local pGui = LocalPlayer:FindFirstChild("PlayerGui")
                if pGui and pGui:FindFirstChild("TouchGui") then
                    local jBtn = pGui.TouchGui:FindFirstChild("JumpButton", true)
                    if jBtn then
                        jBtn.Visible = true
                        jBtn.Active = true
                    end
                end
            end)
        end

        -- 4. HIZ SABİTLEME
        if isSpeedActive and humanoid and humanoid.WalkSpeed ~= targetSpeed then
            humanoid.WalkSpeed = targetSpeed
        end

        -- 5. SKILL WALK
        if skillWalkMode ~= "OFF" and humanoid and hrp then
            local isCasting = (humanoid.WalkSpeed <= 2) or hrp.Anchored or char:FindFirstChild("Stun") or char:FindFirstChild("Skill")
            if (skillWalkMode == "ALWAYS") or (skillWalkMode == "AUTO" and isCasting) then
                if hrp.Anchored then hrp.Anchored = false end
                if humanoid.MoveDirection.Magnitude > 0 then
                    local spd = isSpeedActive and targetSpeed or 16
                    hrp.CFrame = hrp.CFrame + (humanoid.MoveDirection * (spd * delta))
                end
            end
        end

        -- 6. SAFE HITBOX
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local tHrp = player.Character:FindFirstChild("HumanoidRootPart")
                if tHrp then
                    if hitboxActive then
                        tHrp.Size = Vector3.new(safeHitboxSize, safeHitboxSize, safeHitboxSize)
                        tHrp.Transparency = 0.75
                        tHrp.Color = Color3.fromRGB(0, 240, 255)
                        tHrp.Material = Enum.Material.Neon
                        tHrp.CanCollide = false
                    else
                        if tHrp.Size ~= Vector3.new(2, 2, 1) then
                            tHrp.Size = Vector3.new(2, 2, 1)
                            tHrp.Transparency = 1
                        end
                    end
                end
            end
        end

        -- 7. YEŞİL CAN BARLI ESP
        if espActive then
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local tChar = player.Character
                    local head = tChar:FindFirstChild("Head") or tChar:FindFirstChild("HumanoidRootPart")
                    local tHum = tChar:FindFirstChildOfClass("Humanoid")

                    if head and tHum then
                        local hl = tChar:FindFirstChild("LethaESP_Highlight")
                        if not hl then
