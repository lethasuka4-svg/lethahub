-- MOBİL SÜRÜKLEME SİSTEMİ
local function makeDraggable(guiObject)
    local dragging, hasDragged, dragStart, startPos = false, false, nil, nil
    local dragInputObj = nil

    guiObject.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            hasDragged = false
            dragStart = input.Position
            startPos = guiObject.Position
            dragInputObj = input

            local conn
            conn = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    conn:Disconnect()
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input == dragInputObj then
            local delta = input.Position - dragStart
            if delta.Magnitude > 6 then hasDragged = true end
            guiObject.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

    return function() return hasDragged end
end

-- CAMLOCK HEDEF
local function getClosestTarget()
    local nearest = nil
    local shortestDistance = math.huge
    local cam = workspace.CurrentCamera
    if not cam then return nil end
    local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local tHrp = player.Character:FindFirstChild("HumanoidRootPart")
            local tHum = player.Character:FindFirstChildOfClass("Humanoid")
            if tHrp and tHum and tHum.Health > 0 then
                local pos, onScreen = cam:WorldToViewportPoint(tHrp.Position)
                if onScreen then
                    local dist = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                    if dist < shortestDistance then
                        shortestDistance = dist
                        nearest = player
                    end
                end
            end
        end
    end
    return nearest
end

-- HEARTBEAT / RENDER STEPPED KONTROLLERİ
RunService.Heartbeat:Connect(function()
    if LocalPlayer.Character then
        local hrp = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        local hum = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")

        -- KARADAYKEN EN SON GÜVENLİ POZİSYONU KAYDET
        if hum and hrp and hum.FloorMaterial ~= Enum.Material.Air and hum.Health > 0 then
            lastSafeCFrame = hrp.CFrame
        end

        -- ANTI-VOID: BOŞLUĞA DÜŞERSE KARAYA GEL
        if antiVoidActive and hrp and hrp.Position.Y < -30 then
            if lastSafeCFrame then
                hrp.CFrame = lastSafeCFrame + Vector3.new(0, 4, 0)
                hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            end
        end
    end
end)

RunService.RenderStepped:Connect(function(delta)
    if LocalPlayer.Character then
        local char = LocalPlayer.Character
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        local hrp = char:FindFirstChild("HumanoidRootPart")

        -- CAMLOCK
        if camlockActive and camlockSystemEnabled then
            if not camlockTarget or not camlockTarget.Character or not camlockTarget.Character:FindFirstChild("HumanoidRootPart") or (camlockTarget.Character:FindFirstChildOfClass("Humanoid") and camlockTarget.Character:FindFirstChildOfClass("Humanoid").Health <= 0) then
                camlockTarget = getClosestTarget()
            end

            if camlockTarget and camlockTarget.Character and camlockTarget.Character:FindFirstChild("HumanoidRootPart") then
                local tHrp = camlockTarget.Character.HumanoidRootPart
                local predictedPos = tHrp.Position + (tHrp.AssemblyLinearVelocity * predictionAmount)
                local cam = workspace.CurrentCamera
                if cam then
                    cam.CFrame = CFrame.new(cam.CFrame.Position, predictedPos)
                end
            end
        end

        -- HIZ
        if isSpeedActive and humanoid and humanoid.Health > 0 and humanoid.WalkSpeed ~= targetSpeed then
            humanoid.WalkSpeed = targetSpeed
        end

        -- SKILL WALK (STUN KORUMALI)
        if skillWalkMode ~= "OFF" and humanoid and hrp and humanoid.Health > 0 then
            local isStunned = char:FindFirstChild("Stun") or char:FindFirstChild("Knockdown") or (humanoid:GetState() == Enum.HumanoidStateType.Physics)
            local isCasting = (humanoid.WalkSpeed <= 2) or char:FindFirstChild("Skill")

            if not isStunned and ((skillWalkMode == "ALWAYS") or (skillWalkMode == "AUTO" and isCasting)) then
                if humanoid.MoveDirection.Magnitude > 0 then
                    local spd = isSpeedActive and targetSpeed or 16
                    hrp.CFrame = hrp.CFrame + (humanoid.MoveDirection * (spd * delta))
                end
            end
        end

        -- HITBOX & ESP
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local tHrp = player.Character:FindFirstChild("HumanoidRootPart")
                if tHrp then
                    if hitboxActive then
                        tHrp.Size = Vector3.new(safeHitboxSize, safeHitboxSize, safeHitboxSize)
                        tHrp.Transparency = 0.92
                        tHrp.Color = Color3.fromRGB(0, 180, 255)
                        tHrp.Material = Enum.Material.SmoothPlastic
                        tHrp.CanCollide = false
                    else
                        if tHrp.Size ~= Vector3.new(2, 2, 1) then
                            tHrp.Size = Vector3.new(2, 2, 1)
                            tHrp.Transparency = 1
                        end
                    end
                end

                updatePlayerESP(player)
            end
        end
    end
end)
