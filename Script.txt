-- MOBİL / PC SÜRÜKLEME FONKSİYONU
local function makeDraggable(guiObject)
    local dragging, hasDragged, dragStart, startPos = false, false, nil, nil
    local dragInputObj = nil

    guiObject.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            hasDragged = false
            dragStart = input.Position
            startPos = guiObject.Position
            dragInputObj = input

            local conn
            conn = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    conn:Disconnect()
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input == dragInputObj then
            local delta = input.Position - dragStart
            if delta.Magnitude > 6 then hasDragged = true end
            guiObject.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

    return function() return hasDragged end
end

-- HEDEF KONTROLÜ
local function isTargetValid(player)
    if player and player.Parent and player.Character then
        local hum = player.Character:FindFirstChildOfClass("Humanoid")
        local hrp = player.Character:FindFirstChild("HumanoidRootPart")
        if hum and hrp and hum.Health > 0 then
            return true
        end
    end
    return false
end

local function getClosestTarget()
    local nearest, shortestDist = nil, math.huge
    local myChar = LocalPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return nil end
    local myHrp = myChar.HumanoidRootPart

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and isTargetValid(player) then
            local dist = (player.Character.HumanoidRootPart.Position - myHrp.Position).Magnitude
            if dist < shortestDist then
                shortestDist = dist
                nearest = player
            end
        end
    end
    return nearest
end

local function getLowestHealthTarget()
    local lowestTarget, lowestHp = nil, math.huge
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and isTargetValid(player) then
            local hum = player.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health < lowestHp then
                lowestHp = hum.Health
                lowestTarget = player
            end
        end
    end
    return lowestTarget
end

-- TP & HOVER TOGGLE
local function toggleTpHover()
    tpHoverActive = not tpHoverActive
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")

    if tpHoverActive then
        if hrp then
            savedCFrame = hrp.CFrame
            hrp.CFrame = CFrame.new(targetTpPos)
            hrp.Anchored = true
        end
        TpHoverUI.Text = "TP & Hover: ON"
        TpHoverUI.TextColor3 = Color3.fromRGB(0, 240, 255)
        TpHoverStroke.Color = Color3.fromRGB(0, 240, 255)
    else
        if hrp then
            hrp.Anchored = false
            if savedCFrame then
                hrp.CFrame = savedCFrame
            end
        end
        TpHoverUI.Text = "TP & Hover: OFF"
        TpHoverUI.TextColor3 = Color3.fromRGB(255, 60, 80)
        TpHoverStroke.Color = Color3.fromRGB(255, 60, 80)
    end
end

SaveConfigBtn.MouseButton1Click:Connect(function()
    local configData = {
        targetSpeed = targetSpeed, isSpeedActive = isSpeedActive,
        targetJumpPower = targetJumpPower, isJumpActive = isJumpActive,
        skillWalkMode = skillWalkMode, hitboxActive = hitboxActive,
        safeHitboxSize = safeHitboxSize, espActive = espActive,
        infJumpSystemEnabled = infJumpSystemEnabled,
        camlockDistEnabled = camlockDistEnabled,
        camlockHpEnabled = camlockHpEnabled,
        camlockHpTpEnabled = camlockHpTpEnabled,
        tpHoverEnabled = tpHoverEnabled,
        isCoordActive = isCoordActive
    }
    if writefile then
        writefile(configFileName, HttpService:JSONEncode(configData))
        SaveConfigBtn.Text = "Kaydedildi! ✓"
        task.delay(1.5, function() SaveConfigBtn.Text = "Kaydet 💾" end)
    end
end)

LoadConfigBtn.MouseButton1Click:Connect(function()
    if readfile and isfile and isfile(configFileName) then
        local success, data = pcall(function() return HttpService:JSONEncode(readfile(configFileName)) end)
        if success and data then
            targetSpeed = data.targetSpeed or 16
            isSpeedActive = data.isSpeedActive or false
            targetJumpPower = data.targetJumpPower or 50
            isJumpActive = data.isJumpActive or false
            skillWalkMode = data.skillWalkMode or "OFF"
            hitboxActive = data.hitboxActive or false
            espActive = data.espActive or false
            infJumpSystemEnabled = data.infJumpSystemEnabled or false
            camlockDistEnabled = data.camlockDistEnabled or false
            camlockHpEnabled = data.camlockHpEnabled or false
            camlockHpTpEnabled = data.camlockHpTpEnabled or false
            tpHoverEnabled = data.tpHoverEnabled or false
            isCoordActive = data.isCoordActive or false

            LockDistUI.Visible = camlockDistEnabled
            LockHpUI.Visible = camlockHpEnabled
            LockHpTpUI.Visible = camlockHpTpEnabled
            TpHoverUI.Visible = tpHoverEnabled
            CoordUI.Visible = isCoordActive
            InfJumpUI.Visible = infJumpSystemEnabled

            LoadConfigBtn.Text = "Yüklendi! ✓"
            task.delay(1.5, function() LoadConfigBtn.Text = "Yükle 📂" end)
        end
    end
end)

-- ZIPLAMA SPAM
local function executeSpamJump()
    if LocalPlayer.Character then
        local hrp = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        local hum = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if hrp and hum and hum.Health > 0 then
            hrp.AssemblyLinearVelocity = Vector3.new(hrp.AssemblyLinearVelocity.X, targetJumpPower or 52, hrp.AssemblyLinearVelocity.Z)
            hum:ChangeState(Enum.HumanoidStateType.Jumping)
        end
    end
end

InfJumpUI.InputBegan:Connect(function(input)
    if (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1) and not isInfJumpDragged() then
        executeSpamJump()
    end
end)

-- ESP FONKSİYONU
local function updatePlayerESP(player)
    if player ~= LocalPlayer and player.Character then
        local char = player.Character
        local head = char:FindFirstChild("Head")
        local hum = char:FindFirstChildOfClass("Humanoid")

        if head and hum then
            local bg = head:FindFirstChild("LethaESP_Billboard")
            if espActive then
                if not bg then
                    bg = Instance.new("BillboardGui")
                    bg.Name = "LethaESP_Billboard"
                    bg.Adornee = head
                    bg.Size = UDim2.new(0, 110, 0, 42)
                    bg.StudsOffset = Vector3.new(0, 2.8, 0)
                    bg.AlwaysOnTop = true

                    local nameTxt = Instance.new("TextLabel")
                    nameTxt.Name = "NameTxt"
                    nameTxt.Size = UDim2.new(1, 0, 0, 16)
                    nameTxt.BackgroundTransparency = 1
                    nameTxt.TextColor3 = Color3.fromRGB(220, 220, 230)
                    nameTxt.TextSize = 10
                    nameTxt.Font = Enum.Font.GothamBold
                    nameTxt.Parent = bg

                    local hpBg = Instance.new("Frame")
                    hpBg.Name = "HPBg"
                    hpBg.Size = UDim2.new(0.85, 0, 0, 7)
                    hpBg.Position = UDim2.new(0.075, 0, 0.48, 0)
                    hpBg.BackgroundColor3 = Color3.fromRGB(15, 15, 20)
                    hpBg.Parent = bg

                    local hpFill = Instance.new("Frame")
                    hpFill.Name = "HPFill"
                    hpFill.Size = UDim2.new(1, 0, 1, 0)
                    hpFill.BackgroundColor3 = Color3.fromRGB(0, 190, 100)
                    hpFill.BorderSizePixel = 0
                    hpFill.Parent = hpBg

                    local hpTxt = Instance.new("TextLabel")
                    hpTxt.Name = "HPTxt"
                    hpTxt.Size = UDim2.new(1, 0, 1, 0)
                    hpTxt.BackgroundTransparency = 1
                    hpTxt.TextColor3 = Color3.fromRGB(240, 240, 240)
                    hpTxt.TextSize = 8
                    hpTxt.Font = Enum.Font.GothamBold
                    hpTxt.Parent = hpBg

                    bg.Parent = head
                end

                bg.NameTxt.Text = player.DisplayName
                local hpPercent = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
                bg.HPBg.HPFill.Size = UDim2.new(hpPercent, 0, 1, 0)
                bg.HPBg.HPTxt.Text = "%" .. tostring(math.floor(hpPercent * 100))

                local hl = char:FindFirstChild("LethaESP_Highlight")
                if not hl then
                    hl = Instance.new("Highlight")
                    hl.Name = "LethaESP_Highlight"
                    hl.FillColor = Color3.fromRGB(0, 180, 220)
                    hl.OutlineColor = Color3.fromRGB(180, 220, 255)
                    hl.FillTransparency = 0.88
                    hl.Parent = char
                end
            else
                if bg then bg:Destroy() end
                local hl = char:FindFirstChild("LethaESP_Highlight")
                if hl then hl:Destroy() end
            end
        end
    end
end

-- RENDER STEPPED (ANA DÖNGÜ)
RunService.RenderStepped:Connect(function(delta)
    if LocalPlayer.Character then
        local char = LocalPlayer.Character
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        local hrp = char:FindFirstChild("HumanoidRootPart")

        -- KORDİNAT GÜNCELLEME
        if isCoordActive and hrp then
            local pos = hrp.Position
            CoordUI.Text = string.format("X: %.1f | Y: %.1f | Z: %.1f", pos.X, pos.Y, pos.Z)
        end

        -- TP & HOVER (DONDURMA RE-CHECK)
        if tpHoverActive and hrp then
            hrp.CFrame = CFrame.new(targetTpPos)
            hrp.Velocity = Vector3.zero
        end

        -- CAMLOCK 1: EN YAKIN
        if camlockDistActive and camlockDistEnabled then
            if not isTargetValid(camlockDistTarget) then camlockDistTarget = getClosestTarget() end
            if isTargetValid(camlockDistTarget) then
                local tHrp = camlockDistTarget.Character.HumanoidRootPart
                local predictedPos = tHrp.Position + (tHrp.AssemblyLinearVelocity * predictionAmount)
                local cam = workspace.CurrentCamera
                if cam then cam.CFrame = CFrame.new(cam.CFrame.Position, predictedPos) end
            end

        -- CAMLOCK 2: EN AZ CAN
        elseif camlockHpActive and camlockHpEnabled then
            if not isTargetValid(camlockHpTarget) then camlockHpTarget = getLowestHealthTarget() end
            if isTargetValid(camlockHpTarget) then
                local tHrp = camlockHpTarget.Character.HumanoidRootPart
                local predictedPos = tHrp.Position + (tHrp.AssemblyLinearVelocity * predictionAmount)
                local cam = workspace.CurrentCamera
                if cam then cam.CFrame = CFrame.new(cam.CFrame.Position, predictedPos) end
            end

        -- CAMLOCK 3: EN AZ CAN + TP (2 ADIM YANINA)
        elseif camlockHpTpActive and camlockHpTpEnabled then
            if not isTargetValid(camlockHpTpTarget) then camlockHpTpTarget = getLowestHealthTarget() end
            if isTargetValid(camlockHpTpTarget) and hrp then
                local tHrp = camlockHpTpTarget.Character.HumanoidRootPart
                local predictedPos = tHrp.Position + (tHrp.AssemblyLinearVelocity * predictionAmount)
                local cam = workspace.CurrentCamera
                if cam then cam.CFrame = CFrame.new(cam.CFrame.Position, predictedPos) end
                
                -- Rakibin 3 stud (2-3 adım) arkasına/yanına sürekli ışınlanma
                hrp.CFrame = tHrp.CFrame * CFrame.new(0, 0, 3)
            end
        end

        -- HIZ
        if isSpeedActive and humanoid and humanoid.Health > 0 and humanoid.WalkSpeed ~= targetSpeed then
            humanoid.WalkSpeed = targetSpeed
        end

        -- ZIPLAMA GÜCÜ
        if isJumpActive and humanoid and humanoid.Health > 0 then
            humanoid.UseJumpPower = true
            humanoid.JumpPower = targetJumpPower
        end

        -- SKILL WALK
        if skillWalkMode ~= "OFF" and humanoid and hrp and humanoid.Health > 0 then
            local isStunned = char:FindFirstChild("Stun") or char:FindFirstChild("Knockdown") or (humanoid:GetState() == Enum.HumanoidStateType.Physics)
            local isCasting = (humanoid.WalkSpeed <= 2) or char:FindFirstChild("Skill")

            if not isStunned and ((skillWalkMode == "ALWAYS") or (skillWalkMode == "AUTO" and isCasting)) then
                if humanoid.MoveDirection.Magnitude > 0 then
                    local spd = isSpeedActive and targetSpeed or 16
                    hrp.CFrame = hrp.CFrame + (humanoid.MoveDirection * (spd * delta))
                end
            end
        end

        -- HITBOX & ESP
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local tHrp = player.Character:FindFirstChild("HumanoidRootPart")
                if tHrp then
                    if hitboxActive then
                        tHrp.Size = Vector3.new(safeHitboxSize, safeHitboxSize, safeHitboxSize)
                        tHrp.Transparency = 0.94
                        tHrp.Color = Color3.fromRGB(0, 160, 220)
                        tHrp.CanCollide = false
                    else
                        if tHrp.Size ~= Vector3.new(2, 2, 1) then
                            tHrp.Size = Vector3.new(2, 2, 1)
                            tHrp.Transparency = 1
                        end
                    end
                end
                updatePlayerESP(player)
            end
        end
    end
end)
